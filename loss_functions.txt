 351:     def loss(self, targets, preds_dicts, **kwargs):
 352:         """Loss function for BEVDepthHead.
 353: 
 354:         Args:
 355:             gt_bboxes_3d (list[:obj:`LiDARInstance3DBoxes`]): Ground
 356:                 truth gt boxes.
 357:             gt_labels_3d (list[torch.Tensor]): Labels of boxes.
 358:             preds_dicts (dict): Output of forward function.
 359: 
 360:         Returns:
 361:             dict[str:torch.Tensor]: Loss of heatmap and bbox of each task.
 362:         """
 363:         heatmaps, anno_boxes, inds, masks = targets
 364:         return_loss = 0
 365:         for task_id, preds_dict in enumerate(preds_dicts):
 366:             # heatmap focal loss
 367:             preds_dict[0]['heatmap'] = clip_sigmoid(preds_dict[0]['heatmap'])
 368:             num_pos = heatmaps[task_id].eq(1).float().sum().item()
 369:             cls_avg_factor = torch.clamp(reduce_mean(
 370:                 heatmaps[task_id].new_tensor(num_pos)),
 371:                                          min=1).item()
 372:             loss_heatmap = self.loss_cls(preds_dict[0]['heatmap'],
 373:                                          heatmaps[task_id],
 374:                                          avg_factor=cls_avg_factor)
 375:             target_box = anno_boxes[task_id]
 376:             # reconstruct the anno_box from multiple reg heads
 377:             if 'vel' in preds_dict[0].keys():
 378:                 preds_dict[0]['anno_box'] = torch.cat(
 379:                     (preds_dict[0]['reg'], preds_dict[0]['height'],
 380:                      preds_dict[0]['dim'], preds_dict[0]['rot'],
 381:                      preds_dict[0]['vel']),
 382:                     dim=1,
 383:                 )
 384:             else:
 385:                 preds_dict[0]['anno_box'] = torch.cat(
 386:                     (preds_dict[0]['reg'], preds_dict[0]['height'],
 387:                      preds_dict[0]['dim'], preds_dict[0]['rot']),
 388:                     dim=1,
 389:                 )
 390:             # Regression loss for dimension, offset, height, rotation
 391:             num = masks[task_id].float().sum()
 392:             ind = inds[task_id]
 393:             pred = preds_dict[0]['anno_box'].permute(0, 2, 3, 1).contiguous()
 394:             pred = pred.view(pred.size(0), -1, pred.size(3))
 395:             pred = self._gather_feat(pred, ind)
 396:             mask = masks[task_id].unsqueeze(2).expand_as(target_box).float()
 397:             num = torch.clamp(reduce_mean(target_box.new_tensor(num)),
 398:                               min=1e-4).item()
 399:             isnotnan = (~torch.isnan(target_box)).float()
 400:             mask *= isnotnan
 401:             code_weights = self.train_cfg['code_weights']
 402:             bbox_weights = mask * mask.new_tensor(code_weights)
 403:             loss_bbox = self.loss_bbox(pred,
 404:                                        target_box,
 405:                                        bbox_weights,
 406:                                        avg_factor=num)
 407:             return_loss += loss_bbox
 408:             return_loss += loss_heatmap
 409:         return return_loss
