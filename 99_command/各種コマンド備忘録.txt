import csv
from pathlib import Path
from tensorboard.backend.event_processing.event_accumulator import EventAccumulator

log_dir = Path("outputs/bev_depth_dinov3/lightning_logs/version_8")
output = Path("/workspace/BEVDepth/dinov3_loss_detail_v8.csv")

ea = EventAccumulator(str(log_dir), size_guidance={"scalars": 0})
ea.Reload()

rows = []
for tag in ea.Tags().get("scalars", []):
    for event in ea.Scalars(tag):
        rows.append([event.step, tag, event.value])

rows.sort(key=lambda row: (row[0], row[1]))

with output.open("w", newline="") as f:
    writer = csv.writer(f)
    writer.writerow(["step", "tag", "value"])
    writer.writerows(rows)

print("Saved:", output)
print("Rows:", len(rows))
print("Tags:")
for tag in ea.Tags().get("scalars", []):
    print(" -", tag)
PY
